"""Only manifest-declared evaluation assets may enter a participant's data root."""

import hashlib
import io
import json
import tarfile
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch

from nanoprotein import setup_evaluation as setup


def digest(content):
    return hashlib.sha256(content).hexdigest()


class AssetInstallationTests(unittest.TestCase):
    def test_installs_declared_assets_and_discards_unlisted_metadata(self):
        payload = b'{"sequence": "ACD"}'
        source = b'"""Scoring module."""\n'
        manifest = b'{"chain_id":"a"}\n'
        inventory = json.dumps([{"path": "payloads/a.json", "sha256": digest(payload)}]).encode()
        sources = json.dumps({"files": {"scorer.py": digest(source)}, "origin": "private path"}).encode()
        ready = json.dumps({"manifest_path": "CONTACT_MANIFEST.jsonl", "private_note": "secret"}).encode()
        files = {"contact/payloads/a.json": payload, "source/scorer.py": source,
                 "contact/CONTACT_MANIFEST.jsonl": manifest,
                 "contact/PAYLOAD_INVENTORY.json": inventory,
                 "source/SOURCE_MANIFEST.json": sources,
                 "contact/PDB_CONTACT_DATASET_READY.json": ready,
                 "LICENSE": b"license", "ATTRIBUTION.md": b"private transport notes",
                 "source/unlisted.py": b"secret", "extra.txt": b"secret"}
        with tempfile.TemporaryDirectory() as directory:
            root = Path(directory)
            archive = root / "bundle.tar.gz"
            with tarfile.open(archive, "w:gz") as handle:
                for name, content in files.items():
                    info = tarfile.TarInfo(name)
                    info.size = len(content)
                    handle.addfile(info, io.BytesIO(content))
            with patch.multiple(setup, BUNDLE_SHA256=digest(archive.read_bytes()),
                                MANIFEST_SHA256=digest(manifest), INVENTORY_SHA256=digest(inventory),
                                BUNDLE_SOURCE_MANIFEST_SHA256=digest(sources)):
                output = root / "installed"
                extracted_offsets = []
                original_extract = tarfile.TarFile.extractfile

                def observe_read(handle, member):
                    extracted_offsets.append(member.offset)
                    return original_extract(handle, member)

                with patch.object(tarfile.TarFile, "extractfile", observe_read):
                    setup.extract_bundle(archive, output)
                # The first three reads load metadata; all payload reads must be monotonic.
                self.assertEqual(extracted_offsets[3:], sorted(extracted_offsets[3:]))
            self.assertEqual((output / "contact/payloads/a.json").read_bytes(), payload)
            self.assertEqual((output / "source/scorer.py").read_bytes(), source)
            self.assertFalse((output / "source/unlisted.py").exists())
            self.assertFalse((output / "extra.txt").exists())
            self.assertNotIn("origin", json.loads((output / "source/SOURCE_MANIFEST.json").read_text()))
            self.assertNotIn("private_note", json.loads((output / "contact/PDB_CONTACT_DATASET_READY.json").read_text()))
            self.assertNotIn("private", (output / "ATTRIBUTION.md").read_text())

    def test_path_traversal_links_and_duplicates_fail_before_writes(self):
        for name, link, repeat in (("../escape", False, False), ("link", True, False), ("a", False, True)):
            with self.subTest(name=name), tempfile.TemporaryDirectory() as directory:
                root = Path(directory)
                archive = root / "bundle.tar.gz"
                with tarfile.open(archive, "w:gz") as handle:
                    info = tarfile.TarInfo(name)
                    if link:
                        info.type = tarfile.SYMTYPE
                        info.linkname = "../../escape"
                    handle.addfile(info)
                    if repeat:
                        handle.addfile(info)
                with patch.object(setup, "BUNDLE_SHA256", digest(archive.read_bytes())):
                    with self.assertRaises(ValueError):
                        setup.extract_bundle(archive, root / "installed")
                self.assertFalse((root / "installed").exists())


if __name__ == "__main__":
    unittest.main()
